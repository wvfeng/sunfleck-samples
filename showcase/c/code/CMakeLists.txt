cmake_minimum_required(VERSION 3.25)
project(showcase_c_code LANGUAGES C CXX)

if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  include(CTest)
  find_package(Sunfleck 1.0.0 CONFIG REQUIRED)
endif()
add_executable(showcase_c_code main.c layout.c ../../showcase.c)
target_include_directories(showcase_c_code PRIVATE ../..)
set_target_properties(showcase_c_code PROPERTIES WIN32_EXECUTABLE "${WIN32}")
target_link_libraries(showcase_c_code PRIVATE Sunfleck::sunfleck)
target_compile_features(showcase_c_code PRIVATE c_std_11)
if(BUILD_TESTING)
  add_test(NAME showcase_c_code COMMAND showcase_c_code --test)
endif()

if(MSVC)
  target_compile_options(showcase_c_code PRIVATE /utf-8)
endif()
